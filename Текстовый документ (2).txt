<script>
(() => {
  "use strict";

  const LETTERS = "АВЕКМНОРСТУХ";
  const REGIONS = ["77","99","97","177","199","197","777","799","50","150","190","750"];
  const START_BALANCE = 100000;

  const $ = id => document.getElementById(id);

  // --- Состояние ---
  let state = loadState();
  let currentNumber = null;
  let currentPrice = 0;

  function loadState() {
    try {
      const raw = localStorage.getItem("nomerok_state");
      if (raw) return JSON.parse(raw);
    } catch(e) {}
    return {
      id: Math.floor(100000 + Math.random() * 900000),
      balance: START_BALANCE,
      best: "Нет",
      bestPrice: 0,
      theme: "dark"
    };
  }

  function saveState() {
    localStorage.setItem("nomerok_state", JSON.stringify(state));
  }

  function fmt(n) {
    return n.toLocaleString("ru-RU").replace(/,/g, " ");
  }

  // --- Генерация номера ---
  function pick(arr) { return arr[Math.floor(Math.random() * arr.length)]; }

  function genNumber() {
    const l1 = pick(LETTERS.split(""));
    const d = Math.floor(Math.random() * 999) + 1;
    const l2 = pick(LETTERS.split(""));
    const l3 = pick(LETTERS.split(""));
    const region = pick(REGIONS);
    const digits = String(d).padStart(3, "0");
    return { code: l1 + digits + l2 + l3, region };
  }

  function calcPrice(code, region) {
    const digits = code.slice(1, 4);
    const d = parseInt(digits, 10);
    let price = 5000 + Math.floor(Math.random() * 25000);

    if (["77","99","97"].includes(region)) price *= 3;
    else if (["177","199","197"].includes(region)) price *= 2;
    else if (["777","799"].includes(region)) price *= 4;

    const pretty = ["001","007","777","999","666","123","321","111","222",
                    "333","444","555","888","100","200","300","500"];
    if (pretty.includes(digits)) price *= 5;

    const letters = code.replace(/[0-9]/g, "");
    if (letters[0] === letters[1] && letters[1] === letters[2]) price *= 10;
    else if (letters[1] === letters[2]) price *= 2;

    if (d % 100 === 0) price *= 3;
    else if (d % 10 === 0) price *= 1.5;

    return Math.floor(price);
  }

  // --- UI ---
  function render() {
    $("balance").textContent = fmt(state.balance) + " ₽";
    $("cabId").textContent = state.id;
    $("cabBalance").textContent = fmt(state.balance) + " ₽";
    $("cabBest").textContent = state.best;
    $("cabBestPrice").textContent = fmt(state.bestPrice) + " ₽";
    document.body.setAttribute("data-theme", state.theme);
    $("themeBtn").textContent = state.theme === "dark" ? "🌑" : "☀️";
  }

  function toast(msg) {
    const t = $("toast");
    t.textContent = msg;
    t.classList.add("show");
    clearTimeout(t._tm);
    t._tm = setTimeout(() => t.classList.remove("show"), 1600);
  }

  // --- Действия ---
  function roll() {
    const btn = $("rollBtn");
    if (btn.disabled) return;
    btn.disabled = true;

    // Анимация "прокрутки"
    let spins = 0;
    const interval = setInterval(() => {
      const n = genNumber();
      showPlate(n.code, n.region);
      spins++;
      if (spins > 8) {
        clearInterval(interval);
        finishRoll();
        btn.disabled = false;
      }
    }, 60);
  }

  function finishRoll() {
    const n = genNumber();
    currentNumber = n;
    currentPrice = calcPrice(n.code, n.region);
    showPlate(n.code, n.region);
    $("priceLine").innerHTML = "💰 Оценка: <b>" + fmt(currentPrice) + " ₽</b>";
    $("sellBtn").style.display = "block";
    $("sellBtn").textContent = "💵 Продать за " + fmt(currentPrice) + " ₽";
  }

  function showPlate(code, region) {
    $("plate").style.display = "flex";
    $("placeholder").style.display = "none";
    $("plateCode").textContent = code;
    $("plateRegion").textContent = region;
    $("plate").classList.add("spin");
    setTimeout(() => $("plate").classList.remove("spin"), 400);
  }

  function sell() {
    if (!currentNumber) return;
    state.balance += currentPrice;
    let record = false;
    if (currentPrice > state.bestPrice) {
      state.bestPrice = currentPrice;
      state.best = currentNumber.code + " " + currentNumber.region;
      record = true;
    }
    saveState();
    render();

    toast(record ? "🏆 Новый рекорд! +" + fmt(currentPrice) + " ₽"
                 : "✅ Продано +" + fmt(currentPrice) + " ₽");

    currentNumber = null;
    currentPrice = 0;
    $("sellBtn").style.display = "none";
    $("priceLine").innerHTML = "";
    $("plate").style.display = "none";
    $("placeholder").style.display = "block";
  }

  function toggleCabinet() {
    $("cabinet").classList.toggle("show");
  }

  function toggleTheme() {
    state.theme = state.theme === "dark" ? "light" : "dark";
    saveState();
    render();
  }

  function reset() {
    if (!confirm("Сбросить прогресс? Баланс, рекорд и тема вернутся к старту.")) return;
    localStorage.removeItem("nomerok_state");
    state = loadState();
    currentNumber = null;
    currentPrice = 0;
    $("sellBtn").style.display = "none";
    $("priceLine").innerHTML = "";
    $("plate").style.display = "none";
    $("placeholder").style.display = "block";
    render();
    toast("Сброшено");
  }

  // --- Слушатели ---
  $("rollBtn").addEventListener("click", roll);
  $("sellBtn").addEventListener("click", sell);
  $("cabinetBtn").addEventListener("click", toggleCabinet);
  $("themeBtn").addEventListener("click", toggleTheme);
  $("resetBtn").addEventListener("click", reset);

  // --- Старт ---
  render();
})();
</script>
